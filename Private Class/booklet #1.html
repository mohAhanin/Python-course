۱) اعداد

>>> 123 + 222               # جمع صحیح
345
>>> 1.5 * 4                 # ضرب اعشاری
6.0
>>> 2 ** 100                # توان با دقت نامحدود
1267650600228229401496703205376
>>> len(str(2 ** 12345))    # تعداد ارقام یک عدد غول
3717

ماژول‌های عددی

>>> import math
>>> math.pi
3.141592653589793
>>> math.sqrt(85)
9.219544457292887

>>> import random
>>> random.random()          # عدد تصادفی بین ۰ و ۱
0.7082048489415967
>>> random.choice([1, 2, 3, 4])
2

۲) رشته‌ها (str)

ایندکس و اسلایس

>>> S = 'Code'
>>> len(S)
4
>>> S[0]          # ایندکس از صفر
'C'
>>> S[-1]         # آخرین کاراکتر
'e'
>>> S[1:3]        # از ۱ تا قبل از ۳
'od'
>>> S[1:]         # تا انتها
'ode'
>>> S[:3]         # از ابتدا
'Cod'
>>> S[:-1]        # همه جز آخر
'Cod'
>>> S[:]          # کپی کامل
'Code'

عملگرها

>>> S + 'xyz'         # الحاق
'Codexyz'
>>> S * 3             # تکرار
'CodeCodeCode'

تغییرناپذیری (Immutable)

>>> S[0] = 'z'        # خطا!
TypeError: 'str' object does not support item assignment

>>> S = 'Z' + S[1:]   # ساخت رشته‌ی جدید
>>> S
'Zode'

متدهای رشته

>>> S = 'Code'
>>> S.find('od')            # پیدا کردن آفست
1
>>> S.replace('od','abl')   # جایگزینی
'Cable'

>>> line = 'aaa,bbb,ccccc,dd'
>>> line.split(',')
['aaa', 'bbb', 'ccccc', 'dd']

>>> 'code'.upper()
'CODE'
>>> 'code'.isalpha()
True

۳) لیست‌ها (list)

عملگرهای دنباله‌ای

>>> L = [123, 'text', 1.23]
>>> len(L)
3
>>> L[0]
123
>>> L[:-1]
[123, 'text']
>>> L + [4, 5, 6]
[123, 'text', 1.23, 4, 5, 6]
>>> L * 2
[123, 'text', 1.23, 123, 'text', 1.23]

رشد و کوچک شدن (In-place)

>>> L.append('Py')       # افزودن به آخر
>>> L
[123, 'text', 1.23, 'Py']

>>> L.pop(2)             # حذف با ایندکس
1.23
>>> L
[123, 'text', 'Py']

مرتب‌سازی و معکوس (In-place)

>>> M = ['bb', 'aa', 'cc']
>>> M.sort()
>>> M
['aa', 'bb', 'cc']

>>> M.reverse()
>>> M
['cc', 'bb', 'aa']

خطای خارج از بازه

>>> L = [1, 2, 3]
>>> L[99]
IndexError: list index out of range

>>> L[99] = 1
IndexError: list assignment index out of range

تودرتویی – ماتریس ۳×۳

>>> M = [[1, 2, 3],
         [4, 5, 6],
         [7, 8, 9]]

>>> M[1]        # سطر دوم
[4, 5, 6]
>>> M[1][2]     # سطر دوم، ستون سوم
6

۴) دیکشنری‌ها (dict)

>>> D = {'name': 'Pat', 'job': 'dev', 'age': 40}
>>> D['name']
'Pat'
>>> D['job'] = 'mgr'      # تغییر مقدار
>>> D
{'name': 'Pat', 'job': 'mgr', 'age': 40}

ساخت دیکشنری به سه روش

>>> D = {}
>>> D['name'] = 'Pat'
>>> D['job']  = 'dev'
>>> D['age']  = 40

>>> dict(name='Pat', job='dev', age=40)

>>> dict(zip(['name','job','age'], ['Pat','dev',40]))

تودرتویی

>>> rec = {
...     'name': {'first': 'Pat', 'last': 'Smith'},
...     'jobs': ['dev', 'mgr'],
...     'age': 40.5
... }

>>> rec['name']['last']
'Smith'
>>> rec['jobs'][-1]
'mgr'
>>> rec['jobs'].append('janitor')
>>> rec['jobs']
['dev', 'mgr', 'janitor']

کلید ناموجود و راه‌حل‌ها

>>> D = {'a':1, 'b':2, 'c':3}
>>> D['e']
KeyError: 'e'

>>> 'e' in D                    # تست عضویت
False

>>> D.get('a', 'missing')       # با مقدار پیش‌فرض
1
>>> D.get('e', 'missing')
'missing'

>>> D['e'] if 'e' in D else 0   # if/else سه‌گانه
0

پیمایش با حلقه for

>>> D = dict(a=1, b=2, c=3)

>>> list(D.keys())
['a', 'b', 'c']
>>> list(D.values())
[1, 2, 3]
>>> list(D.items())
[('a', 1), ('b', 2), ('c', 3)]

>>> for key in D:
...     print(key, '=>', D[key])
a => 1
b => 2
c => 3

>>> for (k, v) in D.items():
...     print(k, '=>', v)

۵) تاپل‌ها (tuple)

>>> T = (1, 2, 3, 4)
>>> len(T)
4
>>> T + (5, 6)
(1, 2, 3, 4, 5, 6)
>>> T[0], T[1:]
(1, (2, 3, 4))

>>> T.index(4)    # جایگاه ۴
3
>>> T.count(4)    # تعداد تکرار
1

تغییرناپذیری

>>> T[0] = 2
TypeError: 'tuple' object does not support item assignment

>>> T = (2,) + T[1:]      # ساخت تاپل جدید
>>> T
(2, 2, 3, 4)

نکات نحوی

>>> T = 'hack', 3.0, [11, 22, 33]   # پرانتز اختیاری است
>>> T
('hack', 3.0, [11, 22, 33])

>>> T.append(4)
AttributeError: 'tuple' object has no attribute 'append'

>>> one = (5,)   # تاپل تک‌عضوی به کاما نیاز دارد

خواندن متن

>>> f = open('data.txt')
>>> text = f.read()
>>> text
'Hello\nworld!\n'
>>> print(text)
Hello
world!
>>> text.split()
